# Gate 3 input: the Core assemblies that must grant InternalsVisibleTo to each
# listed friend assembly (Android implementations and required Core friends).
#
#   <friend assembly> <- <Core assembly>[, <Core assembly> ...]
#   <friend assembly> <- (none)          (needs no Core internals)
#
# The Android assembly names are fixed by these grants: Core finds its platform
# twin BY NAME (it loads CodeBrix.Android.{Y} and runs its module initializer), so
# an Android assembly must carry exactly the name its Core grants. Every
# CodeBrix.Android.* project under src/ must have a row here.
